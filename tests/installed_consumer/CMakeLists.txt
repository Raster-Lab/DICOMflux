cmake_minimum_required(VERSION 3.24)
project(DICOMfluxInstalledConsumers LANGUAGES C CXX)
find_package(DICOMfluxProbe CONFIG REQUIRED)
enable_testing()
foreach(language c cpp)
  add_executable(consumer_${language} consumer.${language})
  target_link_libraries(consumer_${language} PRIVATE DICOMflux::probe)
  add_test(NAME installed_${language} COMMAND consumer_${language})
endforeach()
set_target_properties(consumer_c PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
set_target_properties(consumer_cpp PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
