cmake_minimum_required(VERSION 3.24)
project(DICOMfluxFoundationConsumers LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(DICOMfluxFoundation CONFIG REQUIRED)
enable_testing()
foreach(language c cpp)
  add_executable(foundation_consumer_${language} consumer.${language})
  target_link_libraries(foundation_consumer_${language} PRIVATE DICOMflux::foundation)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(foundation_consumer_${language} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME installed_foundation_${language} COMMAND foundation_consumer_${language})
endforeach()
