if(NOT DICOMFLUX_SANITIZER MATCHES "^(none|address|undefined|thread)$")
  message(FATAL_ERROR "Unknown sanitizer")
endif()
function(df_qualify_target target)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(NOT DICOMFLUX_SANITIZER STREQUAL "none")
      target_compile_options(${target} PRIVATE -fsanitize=${DICOMFLUX_SANITIZER} -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=${DICOMFLUX_SANITIZER})
    endif()
  endif()
endfunction()
add_library(dicomflux_probe SHARED src/c_api/df0_probe.cpp)
add_library(DICOMflux::probe ALIAS dicomflux_probe)
set_target_properties(dicomflux_probe PROPERTIES EXPORT_NAME probe
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_compile_definitions(dicomflux_probe PRIVATE DICOMFLUX_PROBE_BUILD)
target_include_directories(dicomflux_probe PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
df_qualify_target(dicomflux_probe)
if(UNIX AND NOT APPLE)
  target_link_options(dicomflux_probe PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/cmake/probe.exports")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/docs/architecture/qualification-budgets.json" df_budgets)
set(df_defaults "/* Generated from qualification-budgets.json; DF-0 only. */\n")
foreach(pair "metadata_bytes;METADATA" "attributes;ATTRIBUTES" "declared_bulk_bytes;BULK" "library_tracked_bytes;TRACKED" "transfer_chunk_bytes;CHUNK")
  list(GET pair 0 key)
  list(GET pair 1 macro)
  string(JSON value GET "${df_budgets}" "${key}")
  string(APPEND df_defaults "#define DF_QUAL_${macro} UINT64_C(${value})\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/probe_defaults.h" "${df_defaults}")
